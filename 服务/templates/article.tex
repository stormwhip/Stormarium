% ---- 中文字体：优先 Noto CJK（部署机），其次 Fandol，最后 Windows 字体 ----
% 说明：字体检测用 fontspec 的 \IfFontExistsTF，缺字体时逐级回退，
%       保证「服务器装了 Noto」和「本地只有 Windows 字体」两种环境都能编译。
\IfFontExistsTF{Noto Serif CJK SC}{
  \setCJKmainfont{Noto Serif CJK SC}
  \IfFontExistsTF{Noto Sans CJK SC}{\setCJKsansfont{Noto Sans CJK SC}}{\setCJKsansfont{Noto Serif CJK SC}}
  \IfFontExistsTF{Noto Sans Mono CJK SC}{\setCJKmonofont{Noto Sans Mono CJK SC}}{\setCJKmonofont{Noto Sans CJK SC}}
}{
  \IfFontExistsTF{Noto Sans CJK SC}{
    \setCJKmainfont{Noto Sans CJK SC}
    \setCJKsansfont{Noto Sans CJK SC}
    \setCJKmonofont{Noto Sans CJK SC}
  }{
    \IfFontExistsTF{FandolSong-Regular}{
      \setCJKmainfont{FandolSong-Regular}
      \setCJKsansfont{FandolHei-Regular}
      \setCJKmonofont{FandolFang-Regular}
    }{
      \IfFontExistsTF{SimSun}{
        \setCJKmainfont{SimSun}
        \setCJKsansfont{SimHei}
        \setCJKmonofont{SimSun}
      }{
        \setCJKmainfont{Noto Serif CJK SC}
      }
    }
  }
}
